```python
@check(stages=[1])
def check_rm(cmd: str, stage:int):
    parts = shlex.split(cmd)
    
    if parts[0] == "rm":
        parser = argparse.ArgumentParser(add_help=False)
        parser.add_argument('-r', '--recursive', action='store_true')
        
        args, _ = parser.parse_known_args(parts[1:])

        if args.recursive and '*' in cmd:
            return None

    return False


@command(stages=[2])
def cmd_without_star(cmd: str, stage: int):

    res = {"before": []
            , "cmd": [cmd]
            , "after": []}

    if shlex.split(cmd)[0] == 'rm' and '*' not in cmd:
        res = {"before": ["echo \"А где же звездочка?\""]
                , "cmd": []
                , "after": []}


    return res
```
