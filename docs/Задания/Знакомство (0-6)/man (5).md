Можно было заметить в прошлом задании, что --help выдает подсказки, однако это лишь малая доля информации о команде.

man com

@check(stages=None)
def check_man(cmd: str, stage:int):
    parts = shlex.split(cmd)
    
    if parts[0] == "man":
        parser = argparse.ArgumentParser()
        parser.add_argument('pages', nargs='*')
        
        args, _ = parser.parse_known_args(parts[1:])

        if len(args.pages) == 1 and 'man' in args.pages:
            return True
